# Toolbox external module; Kali pack, 2026-09-14.
# Source: https://www.kali.org/tools/medusa/
# SERVICE_MODULE is an installed Medusa module, such as ssh or ftp; medusa -d lists modules.
# One credential pair is supplied. Authentication affects account lockout counters and may be logged by the service.
# Use Toolbox Dry-Run before execution.
Name: Medusa_Single_Credential
Description: Test one supplied username/password against one selected service with one worker and no connection retries.
Category: Service Enumeration
Command: medusa -h "<<IP>>" -u "<<USERNAME>>" -p "<<PASSWORD>>" -M "<<SERVICE_MODULE>>" -t 1 -T 1 -R 0 -f <<OPTIONS>>

options:
    Custom service port: -n <<PORT>>
    Enable service SSL: -s
