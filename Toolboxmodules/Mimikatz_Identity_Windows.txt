# Use Toolbox Dry-Run before execution.
Name: Mimikatz_Identity_Windows
Description: Run Mimikatz version and current-token identity commands on a Windows SSH host, then exit.
Category: SMB / Windows
Command: python3 "<<PACK_HELPER>>" windows --host "<<WINDOWS_HOST>>" --user "<<SSH_USER>>" --exe "<<WINDOWS_EXE>>" --tool mimikatz <<OPTIONS>>
